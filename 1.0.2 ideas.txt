6. UPDATE: Rebuild the Star-system:
Since my game is a mix of a Bubble Shooter (which values shot efficiency) and Suika (which values huge combos and scores), my star system should reward both playstyles.

Replace my current calculateStars function entirely. Here is the full replacement function to drop into src/game/GameBoard.tsx (or adjust accordingly)

function calculateStars(
  level: (typeof LEVELS)[0],
  score: number,
  shots: number,
  bestCombo: number,
  timeSec: number,
): number {
  let stars = 1; // Star 1: Survival. You reached the target element and beat the level.

  // Star 2: Efficiency (Bubble Shooter skill). 
  // Did you clear the board cleanly without wasting ammo?
  // We use the dynamic starShotsTwo you already calculate in levels.ts
  const efficiencyTarget = level.starShotsTwo ?? (level.parShots ? level.parShots + 5 : 50);
  if (shots <= efficiencyTarget) {
    stars += 1;
  }

  // Star 3: Mastery (Suika skill). 
  // Did you build big chain reactions to multiply your score?
  // We use the dynamic scoreGoal you already calculate in levels.ts
  const masteryTarget = level.scoreGoal ?? (level.targetElement * 1000 * (level.scoreMultiplier ?? 1));
  if (score >= masteryTarget) {
    stars += 1;
  }

  return Math.min(3, stars);
}
